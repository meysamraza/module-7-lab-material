STEP 8. [FORENSIC] DISK ARTIFACTS WITH UTC TIMESTAMPS
======================================================

sudo TZ=UTC stat -c '%y  %n' /mnt/lab2/home/ec2-user/lab2_note.txt /mnt/lab2/home/ec2-user/.bash_history
sudo grep -n "get-caller-identity\|s3 ls" /mnt/lab2/home/ec2-user/.bash_history
sudo tail -10 /mnt/lab2/home/ec2-user/.bash_history

# TZ=UTC makes times match CloudTrail.
# lab2_note.txt time = file creation.
# .bash_history time = session close, not per-command time.
# grep may return lines from earlier sessions (instance reused).
# Separate sessions by the date -u markers and position in the file.
